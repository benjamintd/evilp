find_package(ZLIB REQUIRED)
find_path(GMP_INCLUDE_DIR gmp.h HINTS /opt/homebrew/include)
find_library(GMP_LIBRARY gmp HINTS /opt/homebrew/lib)
if(NOT GMP_INCLUDE_DIR OR NOT GMP_LIBRARY)
  message(FATAL_ERROR "GMP is required (brew install gmp / apt install libgmp-dev)")
endif()
option(LAB_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(LAB_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
