cmake_minimum_required(VERSION 3.20)
project(milp_lab VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(ZLIB REQUIRED)
find_path(GMP_INCLUDE_DIR gmp.h HINTS /opt/homebrew/include)
find_library(GMP_LIBRARY gmp HINTS /opt/homebrew/lib)
if(NOT GMP_INCLUDE_DIR OR NOT GMP_LIBRARY)
  message(FATAL_ERROR "GMP is required (brew install gmp / apt install libgmp-dev)")
endif()
option(LAB_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(LAB_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(lab_common src/common.c)
target_include_directories(lab_common PUBLIC include)
add_executable(milpcheck src/check.c)
target_include_directories(milpcheck SYSTEM PRIVATE ${GMP_INCLUDE_DIR})
target_link_libraries(milpcheck PRIVATE lab_common ${GMP_LIBRARY} ZLIB::ZLIB m)
add_executable(milpbench src/bench.c)
target_link_libraries(milpbench PRIVATE lab_common ZLIB::ZLIB m)
add_library(lab_lp src/model.c src/factor.c src/linear.c src/lp.c src/bb.c)
target_link_libraries(lab_lp PUBLIC lab_common ZLIB::ZLIB m)
add_executable(milpsolve src/solve.c)
target_link_libraries(milpsolve PRIVATE lab_lp)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME checker_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_checker.py $<TARGET_FILE:milpcheck>)
add_test(NAME harness_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_harness.py $<TARGET_FILE:milpbench> $<TARGET_FILE:milpcheck>)
add_test(NAME report_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_report.py)
add_test(NAME lp_certificate_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_lp_cert.py $<TARGET_FILE:milpcheck>)
add_test(NAME lp_solver_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_lp_solver.py $<TARGET_FILE:milpsolve>)
add_test(NAME lp_pipeline_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_lp_pipeline.py $<TARGET_FILE:milpsolve> $<TARGET_FILE:milpcheck>)
add_executable(test_factor tests/test_factor.c)
target_link_libraries(test_factor PRIVATE lab_lp)
add_test(NAME factor_contract COMMAND test_factor)
add_executable(test_reoptimization tests/test_reoptimization.c)
target_link_libraries(test_reoptimization PRIVATE lab_lp)
add_test(NAME reoptimization_contract COMMAND test_reoptimization)
add_test(NAME mip_solver_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_mip_solver.py $<TARGET_FILE:milpsolve> $<TARGET_FILE:milpcheck>)
add_executable(test_linear tests/test_linear.c)
target_link_libraries(test_linear PRIVATE lab_lp)
add_test(NAME linear_refinement_contract COMMAND test_linear)
add_test(NAME numerical_scaling_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_numerics.py $<TARGET_FILE:milpsolve> $<TARGET_FILE:milpcheck>)
